cmake_minimum_required(VERSION 3.21)
project(KwrtStudio VERSION 1.4.2 LANGUAGES C CXX RC)
if(NOT MSVC OR NOT WIN32)
  message(FATAL_ERROR "Build with Visual Studio on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
function(add_bundled_ssh)
  set(archive "${CMAKE_CURRENT_SOURCE_DIR}/third-party-source/libssh2-1.11.1.tar.xz")
  file(SHA256 "${archive}" actual_hash)
  if(NOT actual_hash STREQUAL "9954cb54c4f548198a7cbebad248bdc87dd64bd26185708a294b2b50771e3769")
    message(FATAL_ERROR "Bundled libssh2 source checksum mismatch.")
  endif()
  set(source "${CMAKE_CURRENT_BINARY_DIR}/_deps/libssh2-1.11.1")
  if(NOT EXISTS "${source}/CMakeLists.txt")
    file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/_deps")
  endif()
  set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
  set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
  set(BUILD_STATIC_LIBS ON)
  set(BUILD_SHARED_LIBS OFF)
  set(BUILD_EXAMPLES OFF)
  set(BUILD_TESTING OFF)
  set(CRYPTO_BACKEND WinCNG)
  set(ENABLE_ECDSA_WINCNG ON)
  set(ENABLE_ZLIB_COMPRESSION OFF)
  add_subdirectory("${source}" "${CMAKE_CURRENT_BINARY_DIR}/_deps/ssh-build" EXCLUDE_FROM_ALL)
endfunction()
add_bundled_ssh()
add_executable(KwrtStudio WIN32
  src/main.cpp src/common.cpp src/formats.cpp src/converter.cpp
  src/adapters.cpp src/adapter_ax6000.cpp src/selftest.cpp src/customization.cpp src/network_mode.cpp src/packages.cpp src/package_view.cpp
  src/router_check.cpp src/router_check_tests.cpp src/ssh_client.cpp src/router_view.cpp src/app.rc)
target_include_directories(KwrtStudio PRIVATE vendor src)
target_compile_definitions(KwrtStudio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(KwrtStudio PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_link_options(KwrtStudio PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
target_link_libraries(KwrtStudio PRIVATE user32 gdi32 bcrypt comctl32 comdlg32 ole32 shell32 uxtheme uuid ws2_32 iphlpapi libssh2_static)
add_custom_command(TARGET KwrtStudio POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/tools" "$<TARGET_FILE_DIR:KwrtStudio>/tools")
enable_testing()
add_test(NAME native_selftest COMMAND KwrtStudio --self-test)

install(TARGETS KwrtStudio RUNTIME DESTINATION .)
install(DIRECTORY tools licenses src vendor docs third-party-source DESTINATION .)
install(FILES README.md CHANGELOG.md LICENSE THIRD-PARTY.md CMakeLists.txt profile.example.json DESTINATION .)
